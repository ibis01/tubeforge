    66→function formatDuration(ms: number | null): string {
    67→  if (ms == null) return "—";
    68→  if (ms < 60_000) return `${Math.max(1, Math.round(ms / 1000))}s`;
    69→  return `${(ms / 60_000).toFixed(1)}m`;
    70→}
