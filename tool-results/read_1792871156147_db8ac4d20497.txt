    63→async function api<T>(url: string, opts?: RequestInit): Promise<T> {
    64→  const controller = new AbortController();
    65→  const timer = setTimeout(() => controller.abort(), 5 * 60 * 1000);
    66→  try {
    67→    const res = await fetch(url, {
    68→      ...opts,
    69→      signal: controller.signal,
    70→      headers: { "Content-Type": "application/json", ...(opts?.headers ?? {}) },
    71→    });
    72→    const data = await res.json().catch(() => ({}));
    73→    if (!res.ok) throw new Error((data as { error?: string }).error ?? `HTTP ${res.status}`);
    74→    return data as T;
    75→  } finally {
    76→    clearTimeout(timer);
    77→  }
    78→}
