    62→async function api<T>(url: string, opts?: RequestInit): Promise<T> {
    63→  const controller = new AbortController();
    64→  const timer = setTimeout(() => controller.abort(), 5 * 60 * 1000);
    65→  try {
    66→    const res = await fetch(url, {
    67→      ...opts,
    68→      signal: controller.signal,
    69→      headers: { "Content-Type": "application/json", ...(opts?.headers ?? {}) },
    70→    });
    71→    const data = await res.json().catch(() => ({}));
    72→    if (!res.ok) throw new Error((data as { error?: string }).error ?? `HTTP ${res.status}`);
    73→    return data as T;
    74→  } finally {
    75→    clearTimeout(timer);
    76→  }
    77→}
