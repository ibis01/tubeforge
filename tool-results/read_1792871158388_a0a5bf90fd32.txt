    63→    63→async function api<T>(url: string, opts?: RequestInit): Promise<T> {
    64→    64→  const controller = new AbortController();
    65→    65→  const timer = setTimeout(() => controller.abort(), 5 * 60 * 1000);
    66→    66→  try {
    67→    67→    const res = await fetch(url, {
    68→    68→      ...opts,
    69→    69→      signal: controller.signal,
    70→    70→      headers: { "Content-Type": "application/json", ...(opts?.headers ?? {}) },
    71→    71→    });
    72→    72→    const data = await res.json().catch(() => ({}));
    73→    73→    if (!res.ok) throw new Error((data as { error?: string }).error ?? `HTTP ${res.status}`);
    74→    74→    return data as T;
    75→    75→  } finally {
    76→    76→    clearTimeout(timer);
    77→    77→  }
    78→    78→}
